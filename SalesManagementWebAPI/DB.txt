CREATE PROCEDURE [dbo].[DB_ERROR_LOGGER]

AS
BEGIN
	SELECT  ERROR_NUMBER() AS ErrorNumber,
			ERROR_SEVERITY() AS ErrorSeverity,
			ERROR_STATE() AS ErrorState,
			ERROR_PROCEDURE() AS ErrorProcedure,
			ERROR_LINE() AS ErrorLine,
			ERROR_MESSAGE() AS ErrorMessage;

	declare @i_count bigint ;
	select @i_count = ISNULL(max(ERROR_ID),0) + 1 
	from T_ERROR_LOG
 
	insert into T_ERROR_LOG
			(ERROR_ID,
			ERROR_MSG,
			STACK_TRACE,
			FUNCTION_NAME,
			ERROR_SOURCE,
			USER_ID,
			ERROR_DATE)
	values	(@i_count,
			ERROR_MESSAGE(),
			null,
			ERROR_PROCEDURE(),
			'DataBase',
			null,
			getdate());

END

CREATE   PROCEDURE [dbo].[PKG_AD$get_user_list] --2
(
    @ai_role_id INT = NULL   -- NULL = all roles
)
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        U.USER_ID UserId,
        U.NAME Name,
        U.USERNAME UserName,
        U.EMAIL Email,
        U.MOBILE Mobile,

        -- comma separated roles
        STRING_AGG(C.DESCRIPTION, ', ') 
            WITHIN GROUP (ORDER BY C.DISPLAY_SEQ) AS Role

    FROM dbo.T_USER_MASTER U
    INNER JOIN dbo.T_USER_ROLE UR
        ON U.USER_ID = UR.USER_ID
    INNER JOIN dbo.T_CODE C
        ON C.CODE = CAST(UR.ROLE_ID AS VARCHAR(20))
       AND C.GROUP_NAME = 'ROLE'
       AND C.STATUS = 'A'

    WHERE
        (@ai_role_id IS NULL OR UR.ROLE_ID = @ai_role_id)

    GROUP BY
        U.USER_ID,
        U.NAME,
        U.USERNAME,
        U.EMAIL,
        U.MOBILE,
        U.STATUS

    ORDER BY U.NAME;
END;
GO

CREATE   PROCEDURE [dbo].[PKG_AD$validate_login] 
					   @as_user_name varchar(50),
					   @as_pwd varchar(100),
					   @p_msg varchar(500) OUTPUT,
					   @p_msg_type tinyint OUTPUT, 
					   @p_ret_id bigint OUTPUT
AS 
BEGIN
	BEGIN TRY 	

			IF EXISTS ( SELECT [USER_ID] FROM [dbo].[T_USER_MASTER] WHERE ([USERNAME] = @as_user_name or EMAIL = @as_user_name) and STATUS ='Y' )
			BEGIN			 
					IF EXISTS ( SELECT 1 FROM [dbo].[T_USER_MASTER] 
					WHERE ([USERNAME] = @as_user_name or EMAIL = @as_user_name) AND [PASSWORD] = @as_pwd  )
					BEGIN
						SELECT @p_ret_id = [USER_ID] 
						FROM [dbo].[T_USER_MASTER] 
						WHERE ([USERNAME] = @as_user_name or EMAIL = @as_user_name) AND [PASSWORD] =  @as_pwd  

						SET @p_msg = 'SUCCESS'
						SET @p_msg_type = 2		
					  	RETURN
					END
					ELSE 
					BEGIN
						SET @p_msg = 'Invalid Password....'
						set @p_msg_type = 0
						set @p_ret_id = 0
						return
					END
			END
			ELSE 
			BEGIN
				SET @p_msg = 'Invalid/Inactive User Name..'
				SET @p_msg_type = 0
				SET @p_ret_id = 0
				RETURN
			END
		
	END TRY
	BEGIN CATCH
		EXEC [DB_ERROR_LOGGER]
	END CATCH
END
GO

CREATE   PROCEDURE [dbo].[PKG_ORD$get_order_list]
(
    @ai_user_id    BIGINT,
    @as_order_no  VARCHAR(30) = NULL,
    @ad_from_date  DATE = NULL,
    @ad_to_date    DATE = NULL
)
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        ORDER_ID OrderId,
        ORDER_NO OrderNo,
        om.VENDOR_ID VendorId,
        vm.VENDOR_NAME VendorName,
        vm.VENDOR_CODE VendorCode,
        om.INSERT_DATE OrderDate
    FROM dbo.T_ORDER_MASTER om
    LEFT OUTER JOIN T_VENDOR_MASTER vm ON vm.VENDOR_ID =om.VENDOR_ID 
    WHERE
        vm.INSERT_BY = @ai_user_id
        AND (@as_order_no IS NULL OR om.ORDER_NO LIKE '%' + @as_order_no + '%')
        AND (@ad_from_date IS NULL OR om.INSERT_DATE >= @ad_from_date)
        AND (@ad_to_date IS NULL OR om.INSERT_DATE <= @ad_to_date)
    ORDER BY om.INSERT_DATE DESC;
END;
GO

/****** Object:  StoredProcedure [dbo].[PKG_ORD$save_order_details]    Script Date: 18-02-2026 06:04:26 AM ******/
SET ANSI_NULLS ON

CREATE   PROCEDURE [dbo].[PKG_ORD$save_order_details] 
(
    @ai_order_id   bigint,
    @ai_vendor_id  bigint,
    @t_items       dbo.TT_ORDER_ITEMS readonly,
    @as_ops_mode   varchar(10),             
    @ai_user_id       bigint,
    @p_msg         varchar(500) OUTPUT,
	@p_msg_type    tinyint OUTPUT
)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @ORDER_NO VARCHAR(30);
    DECLARE @NEXT_NO  INT;

    BEGIN TRY
        BEGIN TRAN;
        IF @as_ops_mode IN ('INSERT','UPDATE')
        BEGIN
            -- At least one item required
            IF NOT EXISTS (SELECT 1 FROM @t_items)
            BEGIN
                SET @p_msg = 'Order must contain at least one item.';
                SET @p_msg_type = 0;
                RETURN
            END
            -- Duplicate item check
            IF EXISTS ( SELECT ITEM_ID FROM @t_items GROUP BY ITEM_ID HAVING COUNT(*) > 1 )
            BEGIN
                SET @p_msg = 'Duplicate items found in order item list.';                
                SET @p_msg_type = 0;
                RETURN
            END
        END

        ------------------------------------------------
        -- ADD ORDER
        ------------------------------------------------
        IF @as_ops_mode = 'INSERT'
        BEGIN
            -- Generate ORDER_NO safely
            SELECT
                @NEXT_NO = CAST(PARM_VALUE AS INT) + 1
            FROM dbo.T_SYSTEM_PARM WITH (UPDLOCK, HOLDLOCK)
            WHERE PARM_TYPE = 'ORDER_NO';

            SET @ORDER_NO = CONCAT('ORD-', @NEXT_NO);

            -- Check ORDER_NO already exists
            IF EXISTS ( SELECT 1 FROM dbo.T_ORDER_MASTER WHERE ORDER_NO = @ORDER_NO)
            BEGIN
                SET @p_msg = 'Generated Order No already exists. Please retry.';
                SET @p_msg_type = 0;
                RETURN
            END

            -- Update system parameter
            UPDATE dbo.T_SYSTEM_PARM
            SET
                PARM_VALUE  = CAST(@NEXT_NO AS VARCHAR(100)),
                UPDATE_BY   = CAST(@ai_user_id AS VARCHAR(100)),
                UPDATE_DATE = GETDATE()
            WHERE PARM_TYPE = 'ORDER_NO';
            DECLARE @ORDER_ID BIGINT;

            SELECT @ORDER_ID = ISNULL(MAX(ORDER_ID), 0) + 1
            FROM dbo.T_ORDER_MASTER;

            -- Insert Order Master
            INSERT INTO dbo.T_ORDER_MASTER
            (
                ORDER_ID,
                ORDER_NO,
                VENDOR_ID,
                INSERT_BY,
                INSERT_DATE
            )
            VALUES
            (
                @ORDER_ID,
                @ORDER_NO,
                @ai_vendor_id,
                @ai_user_id,
                GETDATE()
            );

            SET @ai_order_id = @ORDER_ID;

            -- Insert Order Details
            INSERT INTO dbo.T_ORDER_DETAILS
            (
                ORDER_ID,
                ITEM_ID,
                QUANTITY,
                INSERT_BY,
                INSERT_DATE
            )
            SELECT
                @ai_order_id,
                ITEM_ID,
                QUANTITY,
                @ai_user_id,
                GETDATE()
            FROM @t_items;

            
                SET @p_msg = 'Order Placed Succesfully';
                SET @p_msg_type = 2;
        END

        ------------------------------------------------
        -- UPDATE ORDER
        ------------------------------------------------
        ELSE IF @as_ops_mode = 'UPDATE'
        BEGIN
            -- Check order exists
            IF NOT EXISTS (
                SELECT 1 FROM dbo.T_ORDER_MASTER WHERE ORDER_ID = @ai_order_id
            )
            BEGIN
                SET @p_msg = 'Order not found for update.';
                SET @p_msg_type = 0;
            END

            UPDATE dbo.T_ORDER_MASTER
            SET
                VENDOR_ID   = @ai_vendor_id,
                UPDATE_BY   = @ai_user_id,
                UPDATE_DATE = GETDATE()
            WHERE ORDER_ID = @ai_order_id;

            DELETE FROM dbo.T_ORDER_DETAILS
            WHERE ORDER_ID = @ai_order_id;

            INSERT INTO dbo.T_ORDER_DETAILS
            (
                ORDER_ID,
                ITEM_ID,
                QUANTITY,
                INSERT_BY,
                INSERT_DATE
            )
            SELECT
                @ai_order_id,
                ITEM_ID,
                QUANTITY,
                @ai_user_id,
                GETDATE()
            FROM @t_items;
            
            
            SET @p_msg = 'Order Updated Succesfully';
            SET @p_msg_type = 2;
        END

        ------------------------------------------------
        -- DELETE ORDER
        ------------------------------------------------
        ELSE IF @as_ops_mode = 'DELETE'
        BEGIN
            IF NOT EXISTS (
                SELECT 1 FROM dbo.T_ORDER_MASTER WHERE ORDER_ID = @ai_order_id
            )
                SET @p_msg = 'Order not found for delete.';
                SET @p_msg_type = 0;

            DELETE FROM dbo.T_ORDER_DETAILS
            WHERE ORDER_ID = @ai_order_id;

            DELETE FROM dbo.T_ORDER_MASTER
            WHERE ORDER_ID = @ai_order_id;
        END
        ELSE
        BEGIN
            SET @p_msg ='Invalid OPS_MODE.';   
            
                SET @p_msg_type = 0;
        END

        COMMIT TRAN;
    END TRY
    BEGIN CATCH
        SET @p_msg ='Databse Error!!';             
        SET @p_msg_type = 0;                
		EXEC [DB_ERROR_LOGGER]
        ROLLBACK TRAN;        
        THROW;
    END CATCH
END;
GO
